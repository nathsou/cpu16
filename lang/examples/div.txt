
fn div(a: u16, b: u16): u16 {
    if b == 0 { return 65535; }

    var quotient = 0;
    var remainder = 0;
    var i = 15;
   
    while i != 65535 {
        remainder = (remainder << 1) | ((a >> i) & 1);

        if remainder >= b {
            remainder = remainder - b;
            quotient = quotient | (1 << i);
        }

        i = i - 1;
    }

    return quotient;
}

fn main(): u16 {
  return div(1998, 61);
}
