
fn main(): u16 {
    let PPU_REGS = 0xfff0;
    @store(PPU_REGS, 0x8000, 1); // PPU_ADDR = 0x8000 (start of nametable)

    // write "yodl!" to PPU_DATA
    @store(PPU_REGS, 0x79, 2);
    @store(PPU_REGS, 0x6f, 2);
    @store(PPU_REGS, 0x64, 2);
    @store(PPU_REGS, 0x6c, 2);
    @store(PPU_REGS, 0x21, 2);

    return 0;
}
